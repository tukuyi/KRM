defined_text = { 
	name = GetSICName

	text = {
		trigger = {
			any_character = {
				is_second_in_command = yes
				set_temp_variable = {
					temp_trans_sic = var:SICtoken
				}
				PREV = {
					set_temp_variable = {
						gui_sic = var:temp_trans_sic
					}
				}
			}
			# log = "114514 [?gui_sic]"
		}
		localization_key = "[?gui_sic.GetTokenLocalizedKey]"
	}

}

defined_text = { 
	name = GetHOSTittle

	text = {
		trigger = {
			any_character = {
				is_country_leader = yes
				NOT = {
					check_variable = {
						SICtittle = 0
					}
				}
				set_temp_variable = {
					temp_tittle_sic = var:SICtittle
				}
				PREV = {
					set_temp_variable = {
						gui_t_hos = var:temp_tittle_sic
					}
				}
			}
		}
		localization_key = "KRM_sic_tittle_[?gui_t_hos]"
	}

	text = {
		trigger = {}
		localization_key = "KRM_sic_tittle_0"
	}

}
